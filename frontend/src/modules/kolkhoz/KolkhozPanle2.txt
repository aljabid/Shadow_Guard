function uniqueUrls(urls: any[]): string[] {
  return Array.from(
    new Set(
      urls
        .filter(Boolean)
        .map((u) => String(u).trim())
        .filter(Boolean)
    )
  );
}

function normalizeUrl(value: string): string {
  if (value.startsWith("http://") || value.startsWith("https://")) return value;
  if (value.startsWith("@")) return `https://t.me/${value.slice(1)}`;
  if (!value.includes(".") && !value.includes("/")) return `https://t.me/${value}`;
  return `https://${value}`;
}

function getEvidenceUrls(item: any): string[] {
  const sourceData = item?.source_data || item?.source || {};

  const urls = [
    item?.url,
    item?.source_url,
    item?.exchange_url,
    item?.website,
    item?.telegram,
    item?.support_channel,
    sourceData?.source_url,
    ...(item?.evidence_urls || []),
    ...(sourceData?.evidence_urls || []),
    ...(sourceData?.telegram_links || []),
    ...(sourceData?.web_links || []),
    ...(sourceData?.reddit_links || []),
    ...(sourceData?.github_links || []),
    ...(sourceData?.onion_links || []),
  ].filter(Boolean);

  return uniqueUrls(urls.map((u) => normalizeUrl(String(u))));
}

function getHealthStatus(score: number) {
  if (score >= 85) {
    return {
      label: "COLLAPSE RISK",
      icon: "🔴",
      color: "#ef4444",
      confidence: "High",
      probability: Math.min(98, Math.max(85, score - 3)),
    };
  }

  if (score >= 70) {
    return {
      label: "CRITICAL WARNING",
      icon: "🟠",
      color: "#f97316",
      confidence: "High",
      probability: Math.min(84, Math.max(70, score - 5)),
    };
  }

  if (score >= 40) {
    return {
      label: "WATCHLIST",
      icon: "🟡",
      color: "#eab308",
      confidence: "Medium",
      probability: Math.min(69, Math.max(35, score - 8)),
    };
  }

  return {
    label: "HEALTHY",
    icon: "🟢",
    color: "#22c55e",
    confidence: "Medium",
    probability: Math.max(5, Math.min(25, score - 2)),
  };
}

function formatUpdatedTime(item: any): string {
  const value =
    item?.timestamp ||
    item?.updated_at ||
    item?.created_at ||
    item?.source_data?.last_seen ||
    new Date().toISOString();

  try {
    return new Date(value).toLocaleTimeString();
  } catch {
    return new Date().toLocaleTimeString();
  }
}
